
# DFS (Depth First Search)

def DFS(graph, node, visited):
    if node not in visited:
        print(node, end=" ")
        visited.add(node)

        for neighbor in graph[node]:
            DFS(graph, neighbor, visited)


# Graph representation
graph = {
    0: [1, 2],
    1: [3, 4],
    2: [5],
    3: [],
    4: [],
    5: []
}

visited = set()

print("DFS Traversal:")
DFS(graph, 0, visited)
